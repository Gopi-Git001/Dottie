#!/usr/bin/env bash
# DOT-099 FU1 proof at the committed, clean worktree. Counts processes of THIS app's Electron
# binary and smoke supervisors over time after: a forced timeout, kill -9 / SIGTERM / SIGINT of the runner.
set -u
APP=/workspace/dottie-e06/apps/desktop
L=/workspace/smoke-logs/b370fa4
EL="^$APP/node_modules/electron/dist/electron"
cd "$APP"
export PATH=~/.local/node22/bin:$PATH
ts() { date +%Y-%m-%dT%H:%M:%S.%3N%:z; }
cnt() { ps -eo args= | grep -cE "$1" || true; }
snap() { echo "$(ts) [$1] electron=$(cnt "$EL") supervisor=$(cnt '^[^ ]*node [^ ]*scripts/smoke-supervisor\.mjs')"; }
watch_gone() { # $1 label, $2 t0 ns: poll until no electron+supervisor, report ms
  for i in $(seq 1 100); do
    e=$(cnt "$EL"); s=$(cnt '^[^ ]*node [^ ]*scripts/smoke-supervisor\.mjs')
    if [ "$e" = 0 ] && [ "$s" = 0 ]; then echo "$(ts) [$1] none left after $(( ($(date +%s%N)-$2)/1000000 )) ms"; return 0; fi
    sleep 0.1
  done
  echo "$(ts) [$1] STILL ALIVE after 10 s:"; ps -eo pid,pgid,ppid,args | grep -E "$APP|smoke-supervisor" | grep -v grep; return 1
}
echo "commit $(git rev-parse HEAD) porcelain(repo excl docs/evidence)=[$(git -C .. status --porcelain -- .. ':(exclude)docs/evidence' 2>/dev/null)]"
snap start
# 1) forced timeout
node scripts/smoke-run.mjs --runs 1 --timeout-ms 3000 --out $L/forced-timeout --case-id DOT099-FU1-FORCED-TIMEOUT > $L/forced-timeout.console.txt 2>&1; rc=$?; echo "$(ts) [forced] runner exit $rc"
snap forced+0s; sleep 5; snap forced+5s
# 2) kill -9 / SIGTERM / SIGINT of the runner ~3 s into a run
for sig in KILL TERM INT; do
  node scripts/smoke-run.mjs --runs 1 --timeout-ms 90000 --out $L/runner-sig$sig --case-id DOT099-FU1-RUNNER-SIG$sig > $L/runner-sig$sig.console.txt 2>&1 & R=$!
  sleep 3; snap "sig$sig before"
  kill -$sig $R; t0=$(date +%s%N); wait $R; rc=$?; echo "$(ts) [sig$sig] runner exit $rc"
  watch_gone "sig$sig" $t0
  sleep 5; snap "sig$sig +5s"
done
snap end
